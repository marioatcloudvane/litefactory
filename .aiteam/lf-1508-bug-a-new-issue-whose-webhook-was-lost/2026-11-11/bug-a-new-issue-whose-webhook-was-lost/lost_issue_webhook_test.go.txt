package daemon_test

import (
	"slices"
	"testing"
	"time"
)

// A new issue whose `issues opened` webhook never arrived must be picked up by
// the poll within a tick or two (#1508). The issue has a body instruction and
// no comments, so only the poll can name it. This asserts the owner's
// expectation: found within two one-minute heartbeats of being opened.
func TestPollPicksUpNewIssueWhoseWebhookWasLost(t *testing.T) {
	w := newWebhookFixture(t)
	w.clock.mu.Lock()
	w.clock.now = time.Now().UTC().Add(time.Minute)
	w.clock.mu.Unlock()
	origin := recoveryProvider(t, w.fixture)
	configureRecoverySource(t, w, origin)
	w.daemon = newRecoveryDaemon(t, w, origin, "lost-issue-webhook", false)
	w.plantCursor()

	w.clock.advance(10 * time.Second)
	w.assignedIssue("Lost webhook", "/spike investigate the lost webhook")
	for range 2 {
		w.clock.advance(time.Minute)
		w.settle()
	}
	found := slices.Equal(w.turns.triggers(), []int64{0})

	// Control: the next listing, one discovery interval after the last, does
	// find it, so the gap is the listing cadence and not admission.
	w.clock.advance(scheduleInterval)
	w.settle()
	if got := w.turns.triggers(); !slices.Equal(got, []int64{0}) {
		t.Fatalf("even the next listing did not find the issue: executions = %v\nlogs:\n%s", got, w.fixture.logs)
	}
	if !found {
		t.Fatalf("a new issue with a lost webhook was not picked up within two heartbeats; "+
			"it waited for the next listing, %s after the last one", scheduleInterval)
	}
}
